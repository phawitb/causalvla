Loss Function ของ CausalVLA

Ltotal = Ltask + λlatent · Llatent + λaction · Laction + λsmooth · Lsmooth
Lossสูตรความหมาย
L_taskFlow matching loss (SmolVLA เดิม)เรียนรู้ action จาก demonstration
L_latentMSE(z0, zk) สำหรับ k=1..KLatent representation คงที่แม้ภาพเปลี่ยน (causal invariance)
L_actionMSE(a0, ak) สำหรับ k=1..KAction output เหมือนกันแม้ visual input ต่างกัน
L_smoothMSE(at, at+1)ลด action jitter ให้เคลื่อนไหวนุ่มนวล

@PreTrainedConfig.register_subclass("causal_vla")
@dataclass
class CausalVLAConfig(SmolVLAConfig):
    n_counterfactual: int = 3       # K: จำนวนภาพสมมติ
    aug_intensity: float = 1.0

    lambda_latent: float = 0.1      # น้ำหนัก L_latent
    lambda_action: float = 0.1      # น้ำหนัก L_action
    lambda_smooth: float = 0.01     # น้ำหนัก L_smooth

    use_latent_loss: bool = True    # False = Model D
    use_action_loss: bool = True    # False = Model E
modeling_causal_vla.py
class CausalVLAPolicy(SmolVLAPolicy):
    config_class = CausalVLAConfig
    name = "causal_vla"

    def __init__(self, config, **kwargs):
        super().__init__(config, **kwargs)
        from causal_aug import CausalAugmenter
        self.augmenter = CausalAugmenter(K=config.n_counterfactual,
                                          intensity=config.aug_intensity)

    def forward(self, batch, noise=None, time=None, reduction="mean"):
        # 1. L_task (flow matching loss จาก SmolVLA)
        loss_task, loss_dict = super().forward(batch, noise, time, reduction)

        # 2. Extract z_0, a_0
        z_0 = self._extract_latent(batch)
        a_0 = self._extract_action_prediction(batch)

        # 3. Generate K counterfactual images (GPU)
        cf_images = self.augmenter(self._get_images_from_batch(batch))

        # 4. Forward counterfactual → z_k, a_k
        z_ks, a_ks = [], []
        for k in range(self.config.n_counterfactual):
            cf_batch = self._replace_images(batch, cf_images[k])
            z_ks.append(self._extract_latent(cf_batch))
            a_ks.append(self._extract_action_prediction(cf_batch))

        # 5. Invariance losses
        L_latent = sum(F.mse_loss(z_0, z_k) for z_k in z_ks) / len(z_ks)
        L_action = sum(F.mse_loss(a_0, a_k) for a_k in a_ks) / len(a_ks)
        L_smooth = F.mse_loss(a_0[:, :-1, :], a_0[:, 1:, :])

        # 6. Combine
        total_loss = loss_task
        if self.config.use_latent_loss:
            total_loss += self.config.lambda_latent * L_latent
        if self.config.use_action_loss:
            total_loss += self.config.lambda_action * L_action
        total_loss += self.config.lambda_smooth * L_smooth

        return total_loss, loss_dict

def test_causal_augmenter_output_shape():
    augmenter = CausalAugmenter(K=3).to(device)
    obs = torch.randn(2, 3, 256, 256, device=device)
    assert augmenter(obs).shape == (3, 2, 3, 256, 256)

def test_forward_loss_components():
    loss, loss_dict = policy.forward(mock_batch)
    assert not torch.isnan(loss)
    for key in ["loss_task", "loss_latent", "loss_action", "loss_smooth"]:
        assert key in loss_dict

def test_gradient_flow():
    loss, _ = policy.forward(mock_batch)
    loss.backward()
    for name, p in policy.named_parameters():
        if p.requires_grad:
            assert p.grad is not None, f"No gradient: {name}"
